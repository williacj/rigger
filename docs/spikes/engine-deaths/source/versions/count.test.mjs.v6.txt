// ABOUTME: Throwaway tests for #1462's count.mjs, over invented event files only: each ending D1 to
// D7, the selector, unparsed lines, the fields dropped, the extract and its digest gate.
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { createHash, randomUUID } from 'node:crypto';
import { chmodSync, existsSync, mkdtempSync, mkdirSync, readFileSync, readdirSync, statSync, symlinkSync, writeFileSync } from 'node:fs';

const COMMAND = new URL('./count.mjs', import.meta.url).pathname;
const sha = (text) => createHash('sha256').update(text).digest('hex');
const run = (...args) => spawnSync(process.execPath, [COMMAND, ...args], { encoding: 'utf8' });
const dir = () => mkdtempSync(join(tmpdir(), '1462-spike-engineer-'));

/** A stream builder: each event one second after the last, unless `at` moves the clock. */
function stream() {
  const lines = [];
  let clock = Date.parse('2026-11-18T10:00:00.000Z');
  const emit = (run, layer, event, rest = {}) => {
    lines.push(JSON.stringify({ ts: new Date(clock).toISOString(), run, layer, event, ...rest }));
    clock += 1000;
  };
  return { lines, emit, skip: (ms) => { clock += ms; }, text: () => lines.map((line) => `${line}\n`).join('') };
}
const d = () => `d-${randomUUID()}`;
function source(text) {
  const directory = dir();
  writeFileSync(join(directory, 'events.jsonl'), text);
  return directory;
}
/** The value printed on the one line that starts with `label: `. */
function value(output, label) {
  const found = output.split('\n').filter((line) => line.startsWith(`${label}: `));
  assert.equal(found.length, 1, `${label} is printed once in:\n${output}`);
  return found[0].slice(label.length + 2);
}
/** One claim: the pull, the L3 dispatch and the L1 start. */
function begin(s, r, card, id, fields = { role: 'engineer', tier: 'standard', attempt: 1 }) {
  s.emit(r, 'L3', 'pull', { card, kind: 'change', queueDepth: 0, inFlight: 0 });
  s.emit(r, 'L3', 'dispatch', { card, dispatch: id, ...fields });
  s.emit(r, 'L1', 'dispatch.start', { card, dispatch: id, command: '/private/bin/agent', timeout: 60000, workspace: '/private/ws', facts: { a: 'free text' } });
}
const end = (s, r, card, id, fields) => s.emit(r, 'L1', 'dispatch.end', { card, dispatch: id, ...fields });
const release = (s, r, card) => s.emit(r, 'L3', 'slot.release', { card });
const counts = (text) => { const p = run(source(text)); assert.equal(p.status, 0, p.stderr); return p.stdout; };

test('completed: exit 0 then slot.release is no ending', () => {
  const s = stream(); const r = randomUUID(); const id = d();
  begin(s, r, 7, id); end(s, r, 7, id, { exit: 0, ms: 2000 }); release(s, r, 7);
  const out = counts(s.text());
  assert.equal(value(out, 'completed, exit 0'), '1');
  for (const row of ['D1', 'D2', 'D3', 'D4', 'D5']) assert.match(value(out, `${row} dispatches`), /^0/);
  assert.equal(value(out, 'engine deaths, runs with a D1 or D2 dispatch'), '0');
});

const FIELDS = ['ts', 'run', 'layer', 'event', 'card', 'dispatch', 'exit', 'ms', 'role', 'attempt', 'class', 'outcome', 'system', 'concurrency', 'reason', 'unread', 'kept', 'step', 'census', 'maker'];

test('the extract holds the listed fields alone, marks for the text ones, and nothing of a value out of shape', () => {
  const s = stream(); const a = randomUUID(); const id = d();
  begin(s, a, 7, id, { step: 'secret step', attempt: 1 });
  s.emit(a, 'L0', 'group.killed', { card: 7, dispatch: id, group: 5, census: 'secret census' });
  s.emit(a, 'L2', 'attempt.failed', { card: 7, dispatch: id, maker: 'secret-maker', class: 'secret class', outcome: 'secret outcome', role: 'has space secret', env: { KEY: 'secret' }, exit: 'secret', ms: 'secret', system: 'secret' });
  end(s, a, 7, id, { exit: 0, ms: 2, reason: 'secret', unread: 'secret', kept: 'secret', folderLeft: [{ path: '/secret' }] });
  const lines = run(source(s.text()), '--lines').stdout;
  assert.doesNotMatch(lines, /secret|private|free text/);
  for (const line of lines.trimEnd().split('\n')) for (const key of Object.keys(JSON.parse(line))) assert.ok(FIELDS.includes(key), key);
  const last = JSON.parse(lines.trimEnd().split('\n').at(-1));
  assert.deepEqual([last.reason, last.unread, last.kept, last.step, last.census, last.maker], [true, true, true, false, false, false]);
  const out = counts(s.text());
  assert.equal(value(out, 'values dropped as outside their field\'s shape'), '6 (exit: 1, ms: 1, role: 1, class: 1, outcome: 1, system: 1)');
  assert.doesNotMatch(out, /secret|\/private|free text/);
});

test('the archive run writes the checked lines alone, mode 600, and only for the digest passed', () => {
  const s = stream(); const a = randomUUID(); const id = d();
  begin(s, a, 7, id); end(s, a, 7, id, { exit: 137, ms: 5 });
  const from = source(s.text());
  const lines = run(from, '--lines').stdout;
  const digest = value(run(from).stdout, 'private, sha256 of the filtered lines');
  assert.equal(digest, sha(lines));
  const archive = dir(); chmodSync(archive, 0o700);
  const wrong = run(from, archive, '0'.repeat(64));
  assert.equal(wrong.status, 4);
  assert.equal(existsSync(join(archive, 'events.jsonl')), false);
  const p = run(from, archive, digest);
  assert.equal(p.status, 0, p.stderr);
  const file = join(archive, 'events.jsonl');
  assert.deepEqual(readdirSync(archive), ['events.jsonl']);
  assert.equal(readFileSync(file, 'utf8'), lines);
  assert.equal(statSync(file).mode & 0o777, 0o600);
  assert.equal(value(p.stdout, 'private, sha256 of the extract as read back'), digest);
  assert.equal(value(p.stdout, 'extract lines'), '4');
  assert.equal(value(p.stdout, 'private, sha256 of the window\'s lines, before'), sha(s.text()));
  assert.equal(value(p.stdout, 'private, sha256 of the window\'s lines, after'), sha(s.text()));
  assert.equal(run(from, archive, digest).status, 4, 'never overwrites');
  // A judge reruns the count over the extract: the counts are the same, and so is the digest.
  const section = (text) => text.slice(text.indexOf('== counts'), text.indexOf('== private'));
  assert.ok(section(p.stdout).length > 500);
  assert.equal(section(run(archive).stdout), section(p.stdout));
  assert.equal(value(run(archive).stdout, 'private, sha256 of the filtered lines'), digest);
});

/** Runs the archive form with `code` run once, straight after the command's first write in the archive. */
function changing(from, archive, digest, code) {
  const hook = join(dir(), 'hook.mjs');
  writeFileSync(hook, `import fs from 'node:fs'; import { syncBuiltinESMExports } from 'node:module';
    const write = fs.writeFileSync; let done = false;
    fs.writeFileSync = (...args) => { const answer = write(...args); if (!done) { done = true; ${code} } return answer; };
    syncBuiltinESMExports();`);
  return spawnSync(process.execPath, ['--import', hook, COMMAND, from, archive, digest], { encoding: 'utf8' });
}

test('a source that changes between the read and the archive write leaves nothing in the archive', () => {
  const later = JSON.stringify({ ts: '2026-11-25T00:00:00.000Z', run: randomUUID(), layer: 'L3', event: 'trigger' });
  const earlier = JSON.stringify({ ts: '2026-11-20T00:00:00.000Z', run: randomUUID(), layer: 'L3', event: 'trigger' });
  const added = (line) => (from) => `fs.appendFileSync(${JSON.stringify(join(from, 'events.jsonl'))}, ${JSON.stringify(`${line}\n`)});`;
  const moved = (from) => `fs.unlinkSync(${JSON.stringify(join(from, 'events.jsonl'))});`;
  const rotated = (from) => `fs.mkdirSync(${JSON.stringify(join(from, 'n'))}); fs.copyFileSync(${JSON.stringify(join(from, 'events.jsonl'))}, ${JSON.stringify(join(from, 'n', 'events.jsonl'))});`;
  const spoilt = (from, archive) => `for (const each of fs.readdirSync(${JSON.stringify(archive)})) fs.appendFileSync(${JSON.stringify(archive)} + '/' + each, 'x');`;
  for (const [name, change] of [['later', added(later)], ['earlier', added(earlier)], ['moved', moved], ['rotated', rotated], ['spoilt', spoilt]]) {
    const s = stream(); s.emit(randomUUID(), 'L3', 'run.start', { concurrency: 1 });
    const from = source(s.text());
    const digest = value(run(from).stdout, 'private, sha256 of the filtered lines');
    const archive = dir(); chmodSync(archive, 0o700);
    const p = changing(from, archive, digest, change(from, archive));
    assert.equal(p.status, 4, `${name}: ${p.stdout}${p.stderr}`);
    assert.deepEqual(readdirSync(archive), [], name);
    assert.doesNotMatch(p.stderr, /\//, name);
  }
});

test('an extract made while another appeared is not written over it, and no temporary file stays', () => {
  const s = stream(); s.emit(randomUUID(), 'L3', 'run.start', { concurrency: 1 });
  const from = source(s.text());
  const digest = value(run(from).stdout, 'private, sha256 of the filtered lines');
  const archive = dir(); chmodSync(archive, 0o700);
  const p = changing(from, archive, digest, `write(${JSON.stringify(join(archive, 'events.jsonl'))}, 'theirs');`);
  assert.equal(p.status, 4);
  assert.deepEqual(readdirSync(archive), ['events.jsonl']);
  assert.equal(readFileSync(join(archive, 'events.jsonl'), 'utf8'), 'theirs');
});

test('an archive inside the source is refused, and so is one holding the source', () => {
  const s = stream(); s.emit(randomUUID(), 'L3', 'run.start', { concurrency: 1 });
  const top = dir(); chmodSync(top, 0o700);
  const from = join(top, 'source'); mkdirSync(join(from, 'archive'), { recursive: true, mode: 0o700 });
  writeFileSync(join(from, 'events.jsonl'), s.text());
  const digest = value(run(from).stdout, 'private, sha256 of the filtered lines');
  for (const archive of [join(from, 'archive'), from, top]) assert.equal(run(from, archive, digest).status, 4);
  assert.equal(existsSync(join(from, 'archive', 'events.jsonl')), false);
  assert.equal(existsSync(join(top, 'events.jsonl')), false);
});
