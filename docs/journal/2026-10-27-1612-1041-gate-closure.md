The load hook measures imports only: the process-table executable is explicitly protected outside this set.

```sh
node --input-type=module <<'JS'
import { registerHooks } from 'node:module';
import { realpathSync } from 'node:fs';
import { resolve, relative } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
const root = realpathSync.native(process.cwd());
const prefix = pathToFileURL(root + '/').href;
const loaded = new Set();
const hooks = registerHooks({ load(url, context, nextLoad) {
  const result = nextLoad(url, context);
  if (url.startsWith(prefix) && !url.includes('/node_modules/')) loaded.add(relative(root, fileURLToPath(url)));
  return result;
} });
await import(pathToFileURL(resolve(root, 'scripts/gate-check.mjs')));
await import(pathToFileURL(resolve(root, '.github/scripts/self-hosting-check.mjs')));
hooks.deregister();
console.log([...loaded].sort().join('\n'));
JS
```
